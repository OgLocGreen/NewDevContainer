\section{Motivation: KI in der Promotion}

\begin{frame}{Zwei Versagensmodi von LLMs in der Forschung}
  \begin{columns}[T]
    \begin{column}{0.5\textwidth}
      \warn{1. Wissens-Entropie}
      \begin{itemize}
        \item Erkenntnisse verstreuen sich über Dutzende Chats.
        \item Entscheidungen, Begründungen, gelesene Paper gehen verloren.
        \item Kein kumulatives Gedächtnis über Projekte hinweg.

      \begin{itemize}
        \item Erfundene APIs, Flags, Zitate, Fakten.
        \item In der Wissenschaft \hl{inakzeptabel} ---
              eine falsche Referenz untergräbt alles.
      \end{itemize}
    \end{column}
  \end{columns}
  \vfill
  \begin{takeaway}[These]
    Die Antwort ist nicht ,,ein besseres Modell``, sondern \hl{bessere Struktur
    und Guardrails}. Das Template macht KI-Nutzung \emph{diszipliniert} ---
    nicht mächtiger, sondern verlässlicher.
  \end{takeaway}
\end{frame}

\begin{frame}{\code{/sync} \& Drift-Check --- die kleinen Wartungs-Rituale}
\begin{columns}[T,onlytextwidth]

\begin{column}{0.48\textwidth}
\textbf{\code{/sync} --- Zotero $\leftrightarrow$ Obsidian}
\begin{itemize}\scriptsize
  \item \hl{Vertrauenshierarchie:} Zotero = Master für \hl{Metadaten}, Obsidian = Master für \hl{Notizen}.
  \item Diff-basiert, drei Kategorien:
  \begin{itemize}\scriptsize
    \item \good{Neu in Zotero} --- Note anlegen?
    \item \warn{Verwaist in Obsidian} --- nur melden
    \item \hl{Metadata-Drift} --- Felder angleichen?
  \end{itemize}
  \item Pro Kategorie \hl{nachfragen}, erst nach Bestätigung anwenden.
  \item \warn{Nie} Push Obsidian $\rightarrow$ Zotero.
  \item \warn{Nie} automatisches Löschen verwaister Notizen.
\end{itemize}
\end{column}

\begin{column}{0.48\textwidth}
\textbf{Drift-Check --- zyklisch}
\begin{itemize}\scriptsize
  \item Alle \hl{3--4 substantielle Wechsel}, nicht pro Turn.
  \item Kompakter Cross-Check:\\
        \emph{,,wir haben A, B diskutiert, Vault hat X. Drift-Beobachtung: [\ldots]``}
  \item \hl{Frühwarnsystem} --- kein Reflex, kein Nagging.
  \item Bei laufendem Gedankengang: \hl{bis zur nächsten Pause verschieben}.
  \item Ziel: Chat-Realität und Vault-Stand bleiben \good{deckungsgleich}.
\end{itemize}
\end{column}

\end{columns}

\vspace{0.4em}
\begin{takeaway}[Kernidee]
Beide Rituale halten Chat-Realität und Vault-Stand sauber synchron --- ohne den Flow zu unterbrechen.
\end{takeaway}
\end{frame}

\section{Schutzmechanismen}

\begin{frame}{Guardrails, die Vertrauen erzeugen}
  \begin{columns}[T]
    \begin{column}{0.5\textwidth}
      \begin{itemize}
        \item \hl{Chat = ephemeres Arbeitsverzeichnis}, \hl{Vault =
              persistentes Repository} --- bewusste Trennung.
        \item \code{\_DECISIONS.md} ist \hl{append-only}, nie löschen ---
              auditierbare Spur, warum Pfade verlassen wurden.
        \item Nie löschen, sondern nach \code{archive/} verschieben
              (Scratch ist die einzige Ausnahme).
      \end{itemize}
    \end{column}
    \begin{column}{0.5\textwidth}
      \begin{itemize}
        \item \hl{Vertrauenshierarchie} bei \code{/sync}: Zotero = Master
              für Metadaten, Obsidian = Master für Notizen.
        \item \hl{Nie Zitate erfinden} --- explizite Regel im System-Prompt.
        \item \hl{Schreiben nur via /push}, pro Datei bestätigt --- der
              Vault bleibt vertrauenswürdig.
      \end{itemize}
    \end{column}
  \end{columns}
  \vfill
  {\footnotesize Zweisprachig nach Design: \hl{Konversation Deutsch}, aber
  \hl{aller akademischer Inhalt Englisch} (RQs, Titel, Definitionen) ---
  und LaTeX immer als \code{.tex}-Quelle mit Versionsmarken.}
\end{frame}
